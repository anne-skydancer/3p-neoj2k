# The package smoke test, built against the staged package (verify_package.py gives -DSTAGE=...).
cmake_minimum_required(VERSION 3.25)
project(neoj2k_package_smoke LANGUAGES C)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
# NEOJ2K_STATIC_LINK=ON links the static library (neoj2k_static.lib, libneoj2k.a) with NJ2K_STATIC defined.
if(NEOJ2K_STATIC_LINK)
  set(_names neoj2k_static libneoj2k.a)
else()
  set(_names neoj2k)
endif()
find_library(NEOJ2K_LIB NAMES ${_names} PATHS "${STAGE}/lib/release" REQUIRED NO_DEFAULT_PATH)
add_executable(smoke smoke.c)
target_include_directories(smoke PRIVATE "${STAGE}/include")
if(NEOJ2K_STATIC_LINK)
  target_compile_definitions(smoke PRIVATE NJ2K_STATIC)
endif()
target_link_libraries(smoke PRIVATE ${NEOJ2K_LIB})
if(NOT WIN32)
  set_target_properties(smoke PROPERTIES BUILD_RPATH "${STAGE}/lib/release")
endif()
